# pp-document-suite — PINNED dependency set for the SERVICE IMAGE (Dockerfile).
#
# Why this is separate from requirements.txt: that file documents everything the
# suite *can* use, including optional extras (matplotlib, numpy) that only the
# chart-bearing report paths need and that the service image deliberately does
# not install — folding them in would add a large, unused dependency tree to the
# image. This file is exactly what the image installs, fully pinned, so:
#   * builds are reproducible, and
#   * CI can run `pip-audit --strict` against the real deployed set rather than
#     against a range that resolves differently on every run.
#
# Shared entries are kept in lockstep with apps/wwf-docengine/requirements.txt
# and apps/ppdocwiz/requirements.txt, so an advisory fixed in one is fixed in all.
python-docx==1.2.0
# lxml 5.3.0 carried PYSEC-2026-87; 6.1.0 is the fixed line.
lxml==6.1.0
fastapi==0.139.0
starlette==1.3.1
uvicorn[standard]==0.34.0
# Native Word equations (OMML). Optional at the library level, but the service
# image has always installed it, so it is pinned here rather than dropped.
latex2mathml==3.81.0

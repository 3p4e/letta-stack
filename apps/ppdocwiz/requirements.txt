# PP Doc Wiz — pinned runtime dependencies.
#
# These were previously installed unpinned straight in the Dockerfile. That is
# not only a reproducibility problem: the containment of GET /api/download/{name}
# partly rests on Starlette compiling `{name}` to `[^/]+` and uvicorn decoding
# before routing, which is a property of a VERSION, not a promise. Floating those
# versions meant the app's path handling could change under it silently.
#
# Kept in lockstep with apps/wwf-docengine/requirements.txt — same shared web
# stack, so an advisory fixed in one is fixed in both.
fastapi==0.139.0
# starlette pinned explicitly (normally a fastapi transitive) so pip-audit in CI
# sees the exact deployed version — the 0.41.x line carried 8 advisories.
starlette==1.3.1
uvicorn[standard]==0.34.0
pydantic==2.10.4
# The embedded pp-document-suite engine (backend/app.py adds SUITE/scripts to
# sys.path and calls build_from_md + pp_verify in-process).
python-docx==1.2.0
# lxml 5.3.0 carried PYSEC-2026-87; 6.1.0 is the fixed line.
lxml==6.1.0

#   httpx  — the Letta proxy uses stdlib urllib (backend/app.py), so httpx is a
